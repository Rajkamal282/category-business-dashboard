    <script>
        // YAHAN APNA GOOGLE SCRIPT WEB APP URL DALEIN
        const API_URL = 'https://script.google.com/macros/s/AKfycbzaE_VtxHkcRC87Jvl1LYqWfs0teiyyumtBRvI_lyIPiMjFVFrqGchm1_KnASCMpMrw/exec';
        
        let chartInstance = null;

        async function fetchData() {
            document.getElementById('loader').style.display = 'block';
            document.getElementById('dashboardContent').style.display = 'none';
            document.getElementById('chartContent').style.display = 'none';

            try {
                const response = await fetch(API_URL);
                const data = await response.json();
                
                processData(data);
                
                document.getElementById('loader').style.display = 'none';
                document.getElementById('dashboardContent').style.display = 'flex';
                document.getElementById('chartContent').style.display = 'block';
            } catch (error) {
                document.getElementById('loader').innerText = "Data load karne me error aayi. Kripya URL check karein.";
                console.error(error);
            }
        }

        function processData(data) {
            let totalRevenue = 0;
            let totalTraffic = 0;
            let totalSpend = 0;
            let dateLabels = [];
            let revenueData = [];

            // Calculate Revenue & Traffic from Sales Data
            data.sales.forEach(row => {
                let rev = parseFloat(row['Revenue']) || 0;
                let traffic = parseFloat(row['Glance Views']) || 0;
                totalRevenue += rev;
                totalTraffic += traffic;
                
                // For graph (limiting to first 30 points for clean UI)
                if(dateLabels.length < 30) {
                    let d = new Date(row['Date']).toLocaleDateString('en-GB');
                    if(d !== 'Invalid Date') {
                        dateLabels.push(d);
                        revenueData.push(rev);
                    }
                }
            });

            // Calculate Spend from BCG Data
            data.spend.forEach(row => {
                totalSpend += parseFloat(row['Spend']) || 0;
            });

            // Update UI
            document.getElementById('totalRevenue').innerText = '₹ ' + totalRevenue.toLocaleString('en-IN');
            document.getElementById('totalTraffic').innerText = totalTraffic.toLocaleString('en-IN');
            document.getElementById('totalSpend').innerText = '₹ ' + totalSpend.toLocaleString('en-IN');

            // Draw Chart
            drawChart(dateLabels.reverse(), revenueData.reverse());
        }

        function drawChart(labels, data) {
            const ctx = document.getElementById('revenueChart').getContext('2d');
            
            if (chartInstance) {
                chartInstance.destroy();
            }

            chartInstance = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [{
                        label: 'Revenue Trend',
                        data: data,
                        borderColor: '#c65c2b', // Nat habit Orange/Brown color
                        backgroundColor: 'rgba(198, 92, 43, 0.1)',
                        borderWidth: 2,
                        pointBackgroundColor: '#c65c2b',
                        pointRadius: 4,
                        fill: true,
                        tension: 0.3
                    }]
                },
                options: {
                    responsive: true,
                    plugins: { legend: { display: false } },
                    scales: {
                        y: { beginAtZero: true, grid: { color: '#f0f0f0' } },
                        x: { grid: { display: false } }
                    }
                }
            });
        }

        // It will fetch data as soon as the page loads
        window.onload = fetchData;
    </script>
